--1. Retrieve the first and last name of all employees along with their gender and birth date
SELECT FName, LName, Gender, BD
FROM Employee;
--2.List all employees who work in a specific department
SELECT FName, LName
FROM Employee 
WHERE DeptNum = 1;
--3.Display all projects along with their location and city
SELECT PName, Location, City
FROM Project ;
--4.Find the employee who manages each department along with their hire date
SELECT DeptName,FName,LName, HiringDate
FROM Department , Employee 
WHERE Department.SSN = Employee.SSN;
--5.Retrieve all dependents associated with a "specific employee"
SELECT DepName, BD, Gender, SSN
FROM Dependent 
WHERE SSN = 101;
--6.List all employees who work on a "specific project" EX=1, along with their working hours
SELECT  SSN, Hours
FROM WorkingHours 
WHERE PNum = 2;
--7.Display all departments along with their locations
SELECT DeptName, Location
FROM Department, Depet_loc 
WHERE Department.DeptNum = Depet_loc.DeptNum;
--8.Find all employees supervised by a specific employee
SELECT FName,LName
FROM Employee 
WHERE SuperSSN = 101;
--9.Retrieve all projects assigned to a specific department

FROM Project
WHERE DeptNum = 3;
